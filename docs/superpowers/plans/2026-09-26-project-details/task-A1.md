export type ManualProduct = { id: string; name: string; description: string; source?: ProductSource; updatedAt?: number };
export type ManualProductInput = Omit<ManualProduct, "id"> & { id?: string };
export type ProjectBuildRef = { buildId: string; chatId: string | null; version: number; savedAt: number | null };
export type ProjectStep = keyof ManualFlowState;
// ManualProject gains:
//   builds?: ProjectBuildRef[]; lastOpened?: { step: ProjectStep; at: number };
//   showcasedAt?: number | null; cover?: ProductSource | null

export const PROJECT_NAME_MAX = 80;
export const PROJECT_DESC_MAX = 1000;
export const PROJECT_DESC_COUNTER_FROM = 800;
export function newProductId(): string;                    // "prd_" + 8 base-36 chars; attach() uses it
export function normalizeProjects(list: unknown): ManualProject[];
export function keepProductIds(rows: ManualProductInput[], held: ManualProduct[]): ManualProduct[];
// Ctx:
//   updateProject: (id: string, patch: ProjectPatch) => void;              // was Partial<Omit<ManualProject, "id">>

- **The harness:**
  - `npm run test:projects` is the same as `rm -rf .tmp-test && npx tsc -p tests/projects/tsconfig.json && node --test "tests/projects/*.test.mjs"`.
  - `src/lib/<path>.ts(x)` compiles to `.tmp-test/lib/<path>.js`, and a test imports it as `../../.tmp-test/lib/<path>.js`.
  - `include` covers `src/lib/manual/**` and `src/lib/brief/**`, and modules they import are pulled in. A test of a module outside that closure adds the module's folder to `include`.

**Design notes:**
- **Rows without an id on load** take the first free `p<n>`, counting up from their own 1-based position. A stored id is kept when it is a non-empty string that no earlier row holds, so the first row wins. The ids are persisted by the existing save effect (`:227-230`), so they are given once.
- **Rows without an id on write** (the Brief's Step 1 writes `{ name, description }` at `brief-app.tsx:849-858` and `:912-915`) go through `keepProductIds` inside `updateProject`. Each row lines up with a held row by position when the length is unchanged, else by name, and keeps the matched row's id and source. Brief edits therefore never strip identity before the writers task replaces those writes with `mergeProductEdits` (§5.1.7). `mergeProductEdits` hands over rows with ids, and `keepProductIds` keeps those unchanged. This is also why `brief-app.tsx` compiles unchanged once `ManualProduct.id` is required: its rows are `ManualProductInput`, and each gets its id in the store. Stamping `newProductId()` at those two call sites would give every product a new id on every Continue press.
- **Empty `products: []` / `builds: []`** normalize to absent. Most readers treat the two alike (`?.length`, `?? []`), but `network/products.ts:25` reads `[]` as "no products at all", so absent is the safe form.
- **`normalizeProjects(list: unknown)`**: storage that isn't an array reads as `[]`, and entries that aren't objects are dropped. Before, those crashed the hydrate effect.

function productsIn(raw: unknown): ManualProduct[] | undefined {

// COR-86 — the build refs: a string buildId, a chatId that is a string or
// null, an integer version ≥ 1 and a savedAt that is a number or null; a
// build's first ref wins. Never backfilled here — that needs the builds store,
// so buildsOf() reads the legacy links at read time and attach() freezes a

function buildsIn(raw: unknown): ProjectBuildRef[] | undefined {

// A write's product rows, each with an id (COR-87). A row carrying an id no
// earlier row holds keeps it. A row without one is the held row it lines up
// with — by position when the list keeps its length, else by name — and takes
// that row's id, source and updatedAt; with no free match it is a new product
// with a new id. No held id goes to two rows. This is what stops the Brief's
// `{ name, description }` writes stripping identity.
export function keepProductIds(
  rows: ManualProductInput[],
  held: ManualProduct[],
): ManualProduct[] {
  const taken = new Set<string>();
  const own = rows.map((r) => {
    const id = r.id && !taken.has(r.id) ? r.id : null;
    if (id) taken.add(id);
    return id;
  });
  const free = (x: ManualProduct | undefined) =>
    x && !taken.has(x.id) ? x : undefined;
  const byPosition = rows.length === held.length;
  return rows.map((r, i): ManualProduct => {
    let id = own[i];
    let match: ManualProduct | undefined;
    if (id === null) {
      match =
        (byPosition ? free(held[i]) : undefined) ??
        held.find((x) => free(x) && norm(x.name) === norm(r.name));
      id = match?.id ?? newProductId();
      taken.add(id);
    }
    const source = r.source ?? match?.source;
    const updatedAt = r.updatedAt ?? match?.updatedAt;
    return {
      id,
      name: r.name,
      description: r.description,
      ...(source ? { source } : null),
      ...(updatedAt !== undefined ? { updatedAt } : null),
    };
  });
}
